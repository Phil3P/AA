** Ajout en début de fichier : 
;(function(){
  window.__rt_balances = {};
  var origFetch = window.fetch;
  window.fetch = function() {
    return origFetch.apply(this, arguments).then(function(response) {
      var clone = response.clone();
      clone.json().then(function(data) {
        if (data && data.data && data.data.realTokenGnosis && data.data.realTokenGnosis.accountBalances) {
          data.data.realTokenGnosis.accountBalances.forEach(function(b) {
            if (b && b.token && b.token.address) {
              window.__rt_balances[b.token.address.toLowerCase()] = b.amount;
            }
          });
        }
      }).catch(function(){});
      return response;
    });
  };
})();
